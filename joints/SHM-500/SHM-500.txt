# Joint SHM-500 (ШМ-500): reducer DMRD-KB45 + frameless motor RI115-PH KV40
# The motor has no housing/bearings -> the joint gets its own motor housing; the rotor sits on a hub on the reducer input shaft.
JOINT SHM-500
DATA T_nom=500         # N*m, from the name; rated or peak at the output - TO CONFIRM
USE   reducer  DMRD-KB45      components/DMRD/KB45/DMRD-KB45.txt
USE   motor    RI115-PH-KV40  components/T-motor/RI115-PH-KV40/RI115-PH-KV40.txt
USE   sensor   ?              # Lens angle sensor - TO CHOOSE
DESIGN motor_housing   # stator holder on reducer.motor_side; stator OD115 inside
DESIGN rotor_hub       # rotor ID74 on the hub; hub on reducer.input
DESIGN rear_cover      # closes the motor housing, carries the angle sensor
JOIN reducer.motor_side     motor_housing.reducer
JOIN motor.stator           motor_housing.stator
JOIN reducer.input          rotor_hub.reducer
JOIN motor.rotor            rotor_hub.rotor
JOIN motor_housing.rear     rear_cover.housing
OUT  base     reducer.base
OUT  output   reducer.output
END
