# Joint SHM-150 (ШМ-150): reducer DMRD-KB32 + frameless motor RI80-V2-KV75
# The motor has no housing/bearings: the rotor sits on an adapter cup on the reducer input shaft,
# the stator - in a motor housing designed for the joint.
JOINT SHM-150
DATA T_nom=150          # N*m, from the name; rated or peak at the output - TO CONFIRM
#     role     component      file
USE   reducer  DMRD-KB32      components/DMRD/KB32/DMRD-KB32.txt
USE   motor    RI80-V2-KV75   components/T-motor/RI80-V2-KV75/RI80-V2-KV75.txt
# Sensors - not in the joint yet (for now a joint = reducer + motor), space to be kept for:
LATER sensor_out     # angle sensor of the output shaft
LATER sensor_motor   # rotor angle sensor for motor control, driven from the rotor by a belt
# Build in SolidWorks: run BuildFromSpec on this file (reducer + motor bodies + parts + joint assembly).
# The joint frame is the frame of DMRD-KB32 (x=0 - motor end of the input shaft, reducer at x > 0).
PARTS  SHM-150-rotor-cup.txt
PLACE  motor  0  flip            # motor card x=0 (rotor / lamination face) at x=0, Hall side outward (-X)
LINK   SHM-150-01_RotorCup      DMRD-KB32-02_InputShaft
LINK   SHM-150-02_ClampFlange   DMRD-KB32-02_InputShaft
LINK   RI80-V2-KV75-rotor      DMRD-KB32-02_InputShaft   # the rotor turns with the input shaft
# Parts designed for this joint (geometry generated from the ports)
DESIGN rotor_cup       # SHM-150-rotor-cup.txt: tube D36 through the input shaft (centred in both D36 bores), rotor seat D40.5
DESIGN clamp_flange    # SHM-150-rotor-cup.txt: on the output end of the input shaft, 6xM3 D40.5 + 3xM3 into the cup tube
DESIGN motor_housing   # in the PARTS file: on the reducer pilot (base port), stator seat, outer D99 / flange D142
DESIGN rear_cover      # closes the motor housing, carries the angle sensor
#    port A                 port B
JOIN reducer.input          rotor_cup.shaft
JOIN reducer.input          clamp_flange.shaft
JOIN motor.rotor            rotor_cup.rotor
JOIN motor.stator           motor_housing.stator
JOIN motor_housing.rear     rear_cover.housing
# Interfaces of the joint for the arm layout
OUT  base     reducer.base       # TO DECIDE: the motor housing now sits on this pilot - the previous link
#                                  goes on the motor housing flange or through the same bolts
# OUT base   reducer.base       # housing -> link: 122h7 + 12xD5.5 PCD132
OUT  output   reducer.output     # output shaft -> next link: 112h7 + 16xM5 PCD100
END
