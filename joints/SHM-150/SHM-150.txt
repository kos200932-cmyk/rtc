# Joint SHM-150 (ШМ-150): reducer DMRD-KB32 + frameless motor RI80-V2-KV75
# The motor has no housing/bearings: the rotor sits on an adapter cup on the reducer input shaft,
# the stator - in a motor housing designed for the joint.
JOINT SHM-150
DATA T_nom=150          # N*m, from the name; rated or peak at the output - TO CONFIRM
#     role     component      file
USE   reducer  DMRD-KB32      components/DMRD/KB32/DMRD-KB32.txt
USE   motor    RI80-V2-KV75   components/T-motor/RI80-V2-KV75/RI80-V2-KV75.txt
USE   sensor   ?              # Lens angle sensor - TO CHOOSE
# Parts designed for this joint (geometry generated from the ports)
DESIGN rotor_cup       # adapter cup: on reducer.input (45h7, 6xM3 PCD40.5 on the shaft end), carries the rotor ID40.5
DESIGN motor_housing   # stator OD85 holder; fixed to the base link - TO DEFINE
DESIGN rear_cover      # closes the motor housing, carries the angle sensor
#    port A                 port B
JOIN reducer.input          rotor_cup.reducer
JOIN motor.rotor            rotor_cup.rotor
JOIN motor.stator           motor_housing.stator
JOIN motor_housing.rear     rear_cover.housing
# Interfaces of the joint for the arm layout
OUT  base     reducer.base       # housing -> link: 122h7 + 12xD5.5 PCD132
OUT  output   reducer.output     # output shaft -> next link: 112h7 + 16xM5 PCD100
END
