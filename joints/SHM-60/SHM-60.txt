# Joint SHM-60 (ШМ-60): reducer DMRD-KB25 + frameless motor RI70-KV95
# The motor has no housing/bearings: the rotor sits on an adapter cup on the reducer input shaft,
# the stator - in a motor housing designed for the joint.
JOINT SHM-60
DATA T_nom=60          # N*m, from the name; rated or peak at the output - TO CONFIRM
#     role     component      file
USE   reducer  DMRD-KB25      components/DMRD/KB25/DMRD-KB25.txt
USE   motor    RI70-KV95      components/T-motor/RI70-KV95/RI70-KV95.txt
# Sensors - not in the joint yet (for now a joint = reducer + motor), space to be kept for:
LATER sensor_out     # angle sensor of the output shaft
LATER sensor_motor   # rotor angle sensor for motor control, driven from the rotor by a belt
# Build in SolidWorks: run BuildFromSpec on this file (reducer + motor bodies + parts + joint assembly).
# The joint frame is the frame of DMRD-KB25 (x=0 - motor end of the input shaft, reducer at x > 0).
PARTS  SHM-60-parts.txt
PLACE  motor  0  flip            # motor card x=0 (rotor / lamination face) at x=0, Hall side outward (-X)
LINK   SHM-60-01_RotorCup       DMRD-KB25-02_InputShaft
LINK   SHM-60-02_ClampFlange    DMRD-KB25-02_InputShaft
LINK   RI70-KV95-rotor         DMRD-KB25-02_InputShaft   # the rotor turns with the input shaft
# Parts designed for this joint (geometry generated from the ports)
DESIGN rotor_cup       # adapter cup: on reducer.input (38h7, 6xM3 PCD33.5 on the shaft end), carries the rotor ID38
DESIGN motor_housing   # in the PARTS file: on the reducer pilot (base port), stator seat, outer D92 / flange D110
DESIGN rear_cover      # closes the motor housing, carries the angle sensor
#    port A                 port B
JOIN reducer.input          rotor_cup.reducer
JOIN motor.rotor            rotor_cup.rotor
JOIN motor.stator           motor_housing.stator
JOIN motor_housing.rear     rear_cover.housing
# Interfaces of the joint for the arm layout
OUT  base     motor_housing.base   # the reducer mounting repeated on the motor housing flange (same pilot + holes)
OUT  output   reducer.output     # output shaft -> next link: 86h7 + 16xM4 PCD77
END
