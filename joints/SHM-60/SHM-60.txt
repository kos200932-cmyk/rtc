# Joint SHM-60 (ШМ-60): reducer DMRD-KB25 + frameless motor RI70-KV95
# The motor has no housing/bearings: the rotor sits on an adapter cup on the reducer input shaft,
# the stator - in a motor housing designed for the joint.
JOINT SHM-60
DATA T_nom=60          # N*m, from the name; rated or peak at the output - TO CONFIRM
#     role     component      file
USE   reducer  DMRD-KB25      components/DMRD/KB25/DMRD-KB25.txt
USE   motor    RI70-KV95      components/T-motor/RI70-KV95/RI70-KV95.txt
USE   sensor   ?              # Lens angle sensor - TO CHOOSE
# Parts designed for this joint (geometry generated from the ports)
DESIGN rotor_cup       # adapter cup: on reducer.input (38h7, 6xM3 PCD33.5 on the shaft end), carries the rotor ID38
DESIGN motor_housing   # stator OD76 holder; fixed to the base link - TO DEFINE
DESIGN rear_cover      # closes the motor housing, carries the angle sensor
#    port A                 port B
JOIN reducer.input          rotor_cup.reducer
JOIN motor.rotor            rotor_cup.rotor
JOIN motor.stator           motor_housing.stator
JOIN motor_housing.rear     rear_cover.housing
# Interfaces of the joint for the arm layout
OUT  base     reducer.base       # housing -> link: 94h7 + 12xD4.5 PCD102
OUT  output   reducer.output     # output shaft -> next link: 86h7 + 16xM4 PCD77
END
