# SHM-60: adapter cup for the RI70 rotor on the DMRD-KB25 input shaft + clamping flange (level 2: interfaces,
# no chamfers / fillets / standard fasteners). Coordinates: the frame of DMRD-KB25 (x=0 - motor end of
# the input shaft), mm. Same scheme as SHM-150: the 6-M3 of the shaft end are on D33.5, the M3 heads
# (r 14..19.5) would lie under the rotor bore (r 19) - so the cup is a tube through the hollow shaft
# (centred in both D29 bores), clamped by a flange on the output end with the shaft's 6-M3 there.
#   torque: shaft -> 6xM3 -> flange -> 3xM3 -> cup -> rotor (RI70 peak 2.68 N*m)
#   rotor (Hall side outward): x -15..0 on the seat D38; stator lamination x -13..0, windings to x 3.5
#   (3 mm clear of the housing boss D65 at x 6.5), leads outward to x -44.5
#   cables: D20. Flange D38 (= shaft OD) - does not touch the output shaft face (r >= 19).
#   The next link needs a central recess >= D44 x 4 (flange + screw heads, they turn with the input).

PART SHM-60-01_RotorCup
MOTION rotate
P -18   10
P -18   21       # collar D42 - axial stop of the rotor
P -15   21
P -15   19       # rotor seat D38 x 15
P 0     19
P 0     14.5     # shoulder on the shaft face; tube D29 through the shaft
P 59.5  14.5
P 59.5  10       # bore D20 - cables
#       x     dir  n   pcd    size depth  angle
THREAD  59.5  -1   3   24.5   M3   6      0       # 3-M3 in the tube end for the flange
PORT  shaft  0     1   role=fit_in_shaft fit=29g6 fit_x=0..10,37.5..59.5 shoulder_x=0
PORT  rotor  -15   -1  role=rotor_seat fit=38 fit_x=-15..0 collar=42
END

PART SHM-60-02_ClampFlange
MOTION rotate
P 59.5  10
P 59.5  19       # D38
P 62.5  19
P 62.5  10
#      x     dir  n   pcd    d    depth  angle
HOLES  62.5  -1   6   33.5   3.4  0      90       # -> 6-M3 of the shaft end (D33.5)
HOLES  62.5  -1   3   24.5   3.4  0      0        # -> 3-M3 of the cup tube (D24.5)
PORT  shaft  59.5  -1  role=on_shaft_end bolts=6xM3 pcd=33.5 ang=90
END
