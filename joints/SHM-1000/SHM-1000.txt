# Joint SHM-1000 (ШМ-1000): reducer DMRD-80E + actuator AK80-9 V3.0 KV100 (motor + 9:1 planetary + driver)
# The actuator has its own housing and bearings -> it is attached through an adapter plate and a coupling.
# Two stages: total ratio = 9 x ratio(DMRD-80E). AK80-9 has no through bore -> no cable passage along the axis.
JOINT SHM-1000
DATA T_nom=1000        # N*m, from the name; rated or peak at the output - TO CONFIRM
USE   reducer  DMRD-80E          components/DMRD/80E/DMRD-80E.txt
USE   motor    AK80-9-V3-KV100   components/T-motor/AK80-9-V3-KV100/AK80-9-V3-KV100.txt
USE   sensor   ?                 # Lens angle sensor - TO CHOOSE
DESIGN adapter_plate   # AK80-9 front face (D98, 8xM3 on D85) -> reducer.motor_side (recess D110H7 x 6.5, 4xM8 on D145 in the 130 square)
DESIGN coupling        # AK80-9 output (spigot D48, 6xM4 on D28, 2 dowels D3) -> reducer.input (bore D22H7 x 41, key 8, 19 below the face)
JOIN reducer.motor_side     adapter_plate.reducer
JOIN motor.base             adapter_plate.motor
JOIN motor.output           coupling.motor
JOIN reducer.input          coupling.reducer
OUT  base     reducer.base
OUT  output   reducer.output
END
