# LINK-2 (v2, knee): J2 (SHM-1000 output) -> J3 (SHM-KB50 seat) - concept 2026-10-22, robot/RTC-7-layout-v2.txt
# Coordinates: robot world frame at the zero pose, mm. Welded aluminium:
#   plate on the J2 output (disc D190 + web up to the J3 cylinder, 12 thick - clear of the J3 rear cover in the bore),
#   centring skirt on the J2 output D160h7 and its screw holes (as v1),
#   J3 seat cylinder D252 / D185H7 on the J3 axis: face z=293 against the J3 flange (spigot D185h7), 12 x M8 on D230
#   (screws DIN912 from the J3 reducer side); flat x=-14 on the J2 side (3 mm to the J2 housing, x<=-17, while J2 turns);
#   open below (z=192): the cable loop enters the J3 rear from below.
# J2 axis z=137, output face x=0; J3 axis x=105, J3 flange back face z=293, J3 rear (with the cover) z=200.
ASSEMBLY LINK-2
DATA material=AlMg welded concept=2

PART LINK-2-01_Weldment
MOTION fixed
DENSITY 2700
COLOR 175 195 170
# ---- plate on the J2 output
PRISM X 0 12 CIRC 0 137 190
PRISM X 0 12 RECT -100 100 137 285   # web up to the J3 cylinder (below its spigot z 289..293)
PRISM X -13 0 RING 0 137 186 160   # centring skirt, recess 160H7 x 13
# ---- J3 seat cylinder (world Z), centre x=105 y=0
PRISM Z 192 293 RING 105 0 252 185
TAP Z 279 293 105 0 12 230 M8 255   # 12-M8 x 14 from the face (drill ~16): J3 flange, screws from the reducer side
CUT Z 191 294 RECT -60 -14 -130 130   # flat x=-14 (J2 side)
CUT X -28 0 CIRC 0 137 160   # the J2 output shaft D160 (x -27..0) stays clear of the cylinder
CUT X -14 13 CIRC 0 137 60   # centre (J2 output end plate)
CUT X -14 13 PATTERN 0 137 3 142 9 21   # M8 at 69 deg (joint)
CUT X -14 13 PATTERN 0 137 3 142 9 9   # M8 at 81 deg (joint)
CUT X -14 13 PATTERN 0 137 3 142 9 351   # M8 at 99 deg (joint)
CUT X -14 13 PATTERN 0 137 3 110 11 10   # M10 at 80 deg (joint)
CUT X -14 13 PATTERN 0 137 3 110 11 350   # M10 at 100 deg (joint)
CUT X -14 13 PATTERN 0 137 3 142 12 339   # access to the 3 Rc1/8 plugs
PORTP j2_out X 0 -1 role=on_j2_output fit=160H7 bolts=9xM8_D142+6xM10_D110
PORTP j3_base Z 293 1 role=j3_seat fit=185H7 bolts=12xM8_D230 screws_from=reducer_side
END
